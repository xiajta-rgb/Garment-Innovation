# Garment Innovation OS V1 阶段详细技术实现方案
## 一、V1 阶段核心目标与边界定义
### 1.1 核心交付目标（4个月严格交付）
- **技术目标**：完成云边协同基础架构搭建，验证SQLite WASM+OPFS本地存储与Delta增量同步机制，建立Konva.js 2D画布引擎
- **产品目标**：实现**拼积木式快速2D款式组合工具**，支持拖拽式版型拼接、面料一键应用、自动生成基础生产BOM
- **业务目标**：替代设计师80%的Illustrator重复性工作，将单款设计稿绘制时间从4小时缩短至15分钟以内

### 1.2 明确不做的边界（避免范围蔓延）
- ❌ 不实现参数化基因引擎（V2阶段）
- ❌ 不实现WebGL硬件加速面料渲染（V2阶段）
- ❌ 不实现知识图谱与工艺合规校验（V3阶段）
- ❌ 不实现AI生成设计（V4阶段）
- ❌ 不实现多人协同编辑（V3阶段）

### 3.2 边缘数据层实现（核心技术难点）
#### 3.2.1 SQLite WASM + OPFS初始化
```typescript
// src/lib/sqlite/db.ts
import { SQLite3 } from '@sqlite.org/sqlite-wasm';

let db: SQLite3.Database | null = null;

export async function initDatabase(): Promise<SQLite3.Database> {
  if (db) return db;

  // 加载SQLite WASM模块
  const sqlite3 = await SQLite3({
    locateFile: (file: string) => `/sqlite-wasm/${file}`
  });

  // 打开OPFS文件系统中的数据库
  const opfsRoot = await navigator.storage.getDirectory();
  const fileHandle = await opfsRoot.getFileHandle('garment-os-v1.db', { create: true });
  const accessHandle = await fileHandle.createSyncAccessHandle();

  // 创建数据库连接
  db = new sqlite3.Database(accessHandle);

  // 执行初始化SQL脚本
  db.exec(`
    PRAGMA journal_mode=WAL;
    PRAGMA synchronous=NORMAL;
    PRAGMA temp_store=MEMORY;
  `);

  // 创建核心数据表
  createTables(db);

  return db;
}
```

#### 3.3.2 可拖拽组件组实现
```tsx
// src/features/canvas/ComponentGroup.tsx
import { useCallback } from 'react';
import { Group, Path, Transformer } from 'react-konva';
import { useCanvasStore } from '@/store/canvasStore';
import type { CanvasComponent } from '@/types/canvas';

interface Props {
  component: CanvasComponent;
}

export default function ComponentGroup({ component }: Props) {
  const { selectedId, setSelectedId, updateComponent } = useCanvasStore();
  const isSelected = selectedId === component.id;

  const handleDragEnd = useCallback((e: Konva.KonvaEventObject<DragEvent>) => {
    updateComponent(component.id, {
      x: e.target.x(),
      y: e.target.y()
    });
  }, [component.id, updateComponent]);

  const handleTransformEnd = useCallback((e: Konva.KonvaEventObject<TransformEvent>) => {
    const node = e.target;
    updateComponent(component.id, {
      x: node.x(),
      y: node.y(),
      scaleX: node.scaleX(),
      scaleY: node.scaleY(),
      rotation: node.rotation()
    });
  }, [component.id, updateComponent]);

  return (
    <Group
      id={component.id}
      x={component.x}
      y={component.y}
      scaleX={component.scaleX}
      scaleY={component.scaleY}
      rotation={component.rotation}
      draggable
      onClick={() => setSelectedId(component.id)}
      onDragEnd={handleDragEnd}
      onTransformEnd={handleTransformEnd}
    >
      {component.paths.map((path, index) => (
        <Path
          key={index}
          data={path.data}
          fill={path.fill}
          stroke={path.stroke}
          strokeWidth={path.strokeWidth}
        />
      ))}

      {isSelected && (
        <Transformer
          boundBoxFunc={(oldBox, newBox) => {
            // 限制最小尺寸
            if (newBox.width < 50 || newBox.height < 50) {
              return oldBox;
            }
            return newBox;
          }}
        />
      )}
    </Group>
  );
}
```

```tsx
import { useCallback } from 'react';
import { Card } from 'antd';
import { useCanvasStore } from '@/store/canvasStore';
import { useBomStore } from '@/store/bomStore';
import type { Template } from '@/types/template';

interface Props {
  template: Template;
}

export default function TemplateCard({ template }: Props) {
  const { addComponent } = useCanvasStore();
  const { addDefaultBomItems } = useBomStore();

  const handleDragStart = useCallback((e: React.DragEvent) => {
    e.dataTransfer.setData('template', JSON.stringify(template));
  }, [template]);

  const handleDrop = useCallback((e: React.DragEvent) => {
    e.preventDefault();
    const templateData = JSON.parse(e.dataTransfer.getData('template'));
    
    // 添加组件到画布
    addComponent({
      id: `comp-${Date.now()}`,
      templateId: templateData.id,
      name: templateData.name,
      x: 100,
      y: 100,
      scaleX: 1,
      scaleY: 1,
      rotation: 0,
      paths: JSON.parse(templateData.canvasData).paths
    });

    // 添加默认BOM项
    addDefaultBomItems(templateData.defaultBomItems);
  }, [addComponent, addDefaultBomItems]);

  return (
    <Card
      hoverable
      cover={<img src={template.thumbnail} alt={template.name} className="h-24 object-cover" />}
      size="small"
      draggable
      onDragStart={handleDragStart}
      onDragOver={(e) => e.preventDefault()}
      onDrop={handleDrop}
    >
      <Card.Meta title={template.name} />
    </Card>
  );
}
```

```typescript
// src/lib/material/adapters.ts
import type { Material } from '@/types/material';

// 适配器接口
export interface MaterialAdapter {
  fetchMaterials(): Promise<Material[]>;
  searchMaterials(query: string): Promise<Material[]>;
}

// API适配器（对接现有面料库API）
export class ApiMaterialAdapter implements MaterialAdapter {
  constructor(private baseUrl: string, private apiKey: string) {}

  async fetchMaterials(): Promise<Material[]> {
    const response = await fetch(`${this.baseUrl}/api/materials`, {
      headers: { 'Authorization': `Bearer ${this.apiKey}` }
    });
    const data = await response.json();
    return data.map(this.transformMaterial);
  }

  async searchMaterials(query: string): Promise<Material[]> {
    const response = await fetch(`${this.baseUrl}/api/materials/search?q=${query}`, {
      headers: { 'Authorization': `Bearer ${this.apiKey}` }
    });
    const data = await response.json();
    return data.map(this.transformMaterial);
  }

  private transformMaterial(item: any): Material {
    return {
      id: item.id,
      assetCode: item.code,
      name: item.name,
      category: item.category,
      imageUrl: item.image_url,
      widthMm: item.width_mm,
      weightGsm: item.weight_gsm,
      pricePerMeter: item.price_per_meter,
      lossRate: item.loss_rate || 0.12
    };
  }
}

// Excel导入适配器
export class ExcelMaterialAdapter implements MaterialAdapter {
  constructor(private file: File) {}

  async fetchMaterials(): Promise<Material[]> {
    // 使用SheetJS解析Excel文件
    const XLSX = await import('xlsx');
    const data = await this.file.arrayBuffer();
    const workbook = XLSX.read(data);
    const sheet = workbook.Sheets[workbook.SheetNames[0]];
    const json = XLSX.utils.sheet_to_json(sheet);
    
    return json.map((row: any) => ({
      id: `mat-${Date.now()}-${Math.random()}`,
      assetCode: row['物料编码'],
      name: row['物料名称'],
      category: row['类别'],
      imageUrl: row['图片URL'] || '/placeholder-fabric.png',
      widthMm: parseInt(row['幅宽(mm)']),
      weightGsm: parseInt(row['克重(gsm)']),
      pricePerMeter: parseFloat(row['单价(元/米)']),
      lossRate: parseFloat(row['损耗率']) || 0.12
    }));
  }

  async searchMaterials(query: string): Promise<Material[]> {
    const materials = await this.fetchMaterials();
    return materials.filter(m => 
      m.name.toLowerCase().includes(query.toLowerCase()) ||
      m.assetCode.toLowerCase().includes(query.toLowerCase())
    );
  }
}
```

#### 3.5.2 面料应用功能
```typescript
// src/features/canvas/hooks/useMaterialApplication.ts
import { useCallback } from 'react';
import { useCanvasStore } from '@/store/canvasStore';
import { useBomStore } from '@/store/bomStore';
import type { Material } from '@/types/material';

export function useMaterialApplication() {
  const { selectedComponent, updateComponent } = useCanvasStore();
  const { updateBomItemMaterial } = useBomStore();

  const applyMaterialToSelected = useCallback((material: Material) => {
    if (!selectedComponent) return;

    // 创建面料纹理
    const img = new Image();
    img.crossOrigin = 'anonymous';
    img.src = material.imageUrl;

    img.onload = () => {
      const canvas = document.createElement('canvas');
      const ctx = canvas.getContext('2d')!;
      canvas.width = img.width;
      canvas.height = img.height;
      ctx.drawImage(img, 0, 0);

      // 更新组件填充
      const updatedPaths = selectedComponent.paths.map(path => ({
        ...path,
        fill: canvas.toDataURL()
      }));

      updateComponent(selectedComponent.id, { paths: updatedPaths });

      // 更新BOM中的面料信息
      updateBomItemMaterial(selectedComponent.id, material);
    };
  }, [selectedComponent, updateComponent, updateBomItemMaterial]);

  return { applyMaterialToSelected };
}
```

### 3.6 实时BOM生成实现
#### 3.6.1 BOM计算逻辑
```typescript
// src/lib/bom/calculator.ts
import type { CanvasComponent } from '@/types/canvas';
import type { Material } from '@/types/material';

// 计算裁片面积(平方米)
function calculateArea(pathData: string): number {
  // 简化版：使用Konva.js的getClientRect计算边界框面积
  // V2阶段将实现精确的路径面积计算
  const path = new Konva.Path({ data: pathData });
  const rect = path.getClientRect();
  return (rect.width * rect.height) / 1000000; // 转换为平方米
}

// 计算单件面料用量(米)
export function calculateFabricUsage(
  components: CanvasComponent[],
  material: Material
): number {
  let totalArea = 0;
  
  components.forEach(component => {
    component.paths.forEach(path => {
      totalArea += calculateArea(path.data);
    });
  });

  // 面料幅宽(米)
  const fabricWidth = material.widthMm / 1000;
  
  // 理论用量 = 总面积 / 幅宽
  const theoreticalUsage = totalArea / fabricWidth;
  
  // 实际用量 = 理论用量 × (1 + 损耗率)
  const actualUsage = theoreticalUsage * (1 + material.lossRate);

  return Math.round(actualUsage * 1000) / 1000; // 保留3位小数
}

// 计算单件成本
export function calculateCost(usage: number, pricePerMeter: number): number {
  return Math.round(usage * pricePerMeter * 100) / 100; // 保留2位小数
}
```

```typescript
import type { BomItem } from '@/types/bom';

export function exportBomToExcel(bomItems: BomItem[], styleCode: string) {
  // 转换为Excel数据格式
  const data = bomItems.map(item => ({
    '物料编码': item.material.assetCode,
    '物料名称': item.material.name,
    '使用部位': item.componentName,
    '规格': `${item.material.widthMm}mm / ${item.material.weightGsm}gsm`,

    '单位': item.unit,
    '单价(元)': item.material.pricePerMeter,
    '金额(元)': item.cost,
    '损耗率': `${item.material.lossRate * 100}%`
  }));

  // 添加合计行
  const totalCost = bomItems.reduce((sum, item) => sum + item.cost, 0);
  data.push({
    '物料编码': '',
    '物料名称': '合计',
    '使用部位': '',
    '规格': '',
    '单件用量': '',
    '单位': '',
    '单价(元)': '',
    '金额(元)': totalCost,
    '损耗率': ''
  });

  // 创建工作簿
  const worksheet = XLSX.utils.json_to_sheet(data);
  const workbook = XLSX.utils.book_new();
  XLSX.utils.book_append_sheet(workbook, worksheet, 'BOM');

  // 设置列宽
  worksheet['!cols'] = [
    { wch: 15 }, { wch: 20 }, { wch: 15 }, { wch: 20 },
    { wch: 10 }, { wch: 8 }, { wch: 12 }, { wch: 10 }, { wch: 10 }
  ];

  // 导出文件
  const excelBuffer = XLSX.write(workbook, { bookType: 'xlsx', type: 'array' });
  const blob = new Blob([excelBuffer], { type: 'application/octet-stream' });
  saveAs(blob, `${styleCode}-BOM.xlsx`);
}
```

### 4.3 云边增量同步机制
```typescript
// src/lib/sync/delta-sync.ts
import { diff } from 'jsondiffpatch';
import type { EdgeGarment } from '@/types/garment';

// 生成两个版本之间的Delta补丁
export function generateDelta(oldGarment: EdgeGarment, newGarment: EdgeGarment): any {
  return diff(oldGarment, newGarment);
}

// 应用Delta补丁到旧版本
export function applyDelta(oldGarment: EdgeGarment, delta: any): EdgeGarment {
  return {
    ...oldGarment,
    ...delta,
    version: oldGarment.version + 1,
    updatedAt: Date.now()
  };
}

// 同步服务
export class SyncService {
  async syncGarment(localGarment: EdgeGarment, userId: string) {
    // 1. 获取云端最新版本
    const cloudGarment = await this.fetchCloudGarment(localGarment.id, userId);

    if (!cloudGarment) {
      // 云端不存在，直接上传
      return this.uploadGarment(localGarment, userId);
    }

    if (localGarment.version > cloudGarment.version) {
      // 本地版本更新，生成Delta并上传
      const delta = generateDelta(cloudGarment, localGarment);
      return this.uploadDelta(localGarment.id, delta, userId);
    } else if (localGarment.version < cloudGarment.version) {
      // 云端版本更新，下载Delta并应用
      const delta = await this.fetchDelta(localGarment.id, localGarment.version, userId);
      return applyDelta(localGarment, delta);
    }

    // 版本相同，无需同步
    return localGarment;
  }

}
```
