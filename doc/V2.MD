# Garment Innovation OS V2 阶段开发文档
## 一、V2 阶段核心定位与目标
### 1.1 战略定位
V2 是 Garment Innovation OS 从"**拼积木工具**"向"**工业级参数化设计平台**"跃迁的关键里程碑。它彻底解决 V1 阶段"静态图形缩放导致版型畸变"的核心痛点，建立"**参数驱动几何、几何驱动成本**"的闭环数据链路，为 V3 阶段的 Garment DSL 全面落地和 V4 阶段的 AI 生成奠定不可替代的底层基础。

### 1.2 核心交付目标（6个月严格交付）
- **技术目标**：
  - 上线**参数化基因引擎**，实现无畸变的版型参数化调整
  - 部署**Konva.js + PixiJS 双核混合渲染管线**，解决复杂面料性能瓶颈
  - 建立**本地规则引擎**，实现基础工艺约束校验
  - 完成**实时数字孪生 BOM 成本沙盘**，实现毫秒级成本更新
- **产品目标**：
  - 设计师滑动滑块即可自动推导符合工艺标准的版型变化
  - 支持 100+ 可调整参数，覆盖 90% 以上的工装品类设计需求
  - 面料渲染效果达到印刷级精度，4K 屏幕下稳定 60FPS
- **业务目标**：
  - 将单款设计修改时间从 2 小时缩短至 5 分钟
  - 打版师介入率从 80% 降低至 20%
  - 成本估算准确率从 70% 提升至 95%

### 1.3 明确不做的边界（避免范围蔓延）
- ❌ 不实现完整 Garment DSL 编译器（V3 阶段）
- ❌ 不实现知识图谱与复杂工艺合规推理（V3 阶段）
- ❌ 不实现多人实时协同编辑（V3 阶段）
- ❌ 不实现 AI 生成设计（V4 阶段）
- ❌ 不实现自动放码（V2.5 迭代）

#### 3.1.2 DAG 有向无环图几何解算器
```typescript
// src/lib/parametric/dag-solver.ts
import { evaluate } from 'mathjs';
import { Bezier } from 'bezier-js';
import type { Gene, GeneInstance } from '@/types/gene';

export class DAGSolver {
  private geneInstances: Map<string, GeneInstance> = new Map();
  private dependencyGraph: Map<string, string[]> = new Map();

  // 注册基因实例
  registerGeneInstance(instance: GeneInstance) {
    this.geneInstances.set(instance.id, instance);
    
    // 构建依赖图
    const dependencies = instance.gene.parameters
      .filter(p => p.dependsOn)
      .map(p => p.dependsOn!.parameterId);
    
    this.dependencyGraph.set(instance.id, dependencies);
  }

  // 解算单个基因实例的几何形状
  solveGeneInstance(instanceId: string): PathData[] {
    const instance = this.geneInstances.get(instanceId);
    if (!instance) throw new Error(`Gene instance ${instanceId} not found`);

    // 1. 先解算所有依赖的参数
    this.solveDependencies(instanceId);

    // 2. 执行几何生成函数
    const generator = new Function('params', 'Bezier', instance.gene.geometryGenerator);
    const paths = generator(instance.parameterValues, Bezier);

    // 3. 应用工艺约束校验
    this.validateConstraints(instance, paths);

    return paths;
  }

  // 递归解算依赖参数
  private solveDependencies(instanceId: string) {
    const dependencies = this.dependencyGraph.get(instanceId) || [];
    dependencies.forEach(depId => {
      this.solveGeneInstance(depId);
    });
  }

  // 工艺约束校验
  private validateConstraints(instance: GeneInstance, paths: PathData[]) {
    instance.gene.constraints.forEach(rule => {
      const context = {
        params: instance.parameterValues,
        paths: paths
      };
      
      const isValid = evaluate(rule.expression, context);
      if (!isValid) {
        if (rule.level === 'error') {
          throw new Error(rule.errorMessage);
        } else {
          console.warn(rule.errorMessage);
        }
      }
    });
  }
}
```

```tsx
// src/features/canvas/HybridRenderer.tsx
import { useRef, useEffect, useState } from 'react';
import { Stage, Layer } from 'react-konva';
import * as PIXI from 'pixi.js';
import { useCanvasStore } from '@/store/canvasStore';

export default function HybridRenderer() {
  const konvaStageRef = useRef<Konva.Stage>(null);
  const pixiCanvasRef = useRef<HTMLCanvasElement>(null);
  const pixiAppRef = useRef<PIXI.Application | null>(null);
  const { components, stageWidth, stageHeight } = useCanvasStore();

  // 初始化 PixiJS 应用
  useEffect(() => {
    if (!pixiCanvasRef.current) return;

    pixiAppRef.current = new PIXI.Application({
      view: pixiCanvasRef.current,
      width: stageWidth,
      height: stageHeight,
      transparent: true,
      antialias: true,
      resolution: window.devicePixelRatio || 1
    });

    return () => {
      pixiAppRef.current?.destroy();
    };
  }, [stageWidth, stageHeight]);

  // 同步 Konva 路径到 PixiJS 渲染
  useEffect(() => {
    if (!pixiAppRef.current || !konvaStageRef.current) return;

    const app = pixiAppRef.current;
    app.stage.removeChildren();

    components.forEach(component => {
      component.paths.forEach(path => {
        // 1. 从 Konva 路径创建 PixiJS Graphics
        const graphics = new PIXI.Graphics();
        
        // 解析 SVG 路径数据并绘制
        graphics.beginPath();
        graphics.parsePath(path.data);
        graphics.closePath();
        
        // 2. 应用面料纹理
        if (path.fill && path.fill.startsWith('http')) {
          const texture = PIXI.Texture.from(path.fill);
          texture.baseTexture.wrapMode = PIXI.WRAP_MODES.REPEAT;
          
          const sprite = new PIXI.Sprite(texture);
          sprite.mask = graphics;
          app.stage.addChild(sprite);
        } else {
          // 纯色填充
          graphics.beginFill(PIXI.Color.shared.setValue(path.fill || '#ffffff'));
          graphics.endFill();
        }
        
        // 3. 绘制描边
        graphics.lineStyle(path.strokeWidth || 1, PIXI.Color.shared.setValue(path.stroke || '#000000'));
        graphics.beginPath();
        graphics.parsePath(path.data);
        graphics.closePath();
        
        app.stage.addChild(graphics);
      });
    });

    app.render();
  }, [components]);

  return (
    <div className="relative" style={{ width: stageWidth, height: stageHeight }}>
      {/* PixiJS 渲染层（底层，负责面料纹理） */}
      <canvas
        ref={pixiCanvasRef}
        className="absolute top-0 left-0"
        style={{ zIndex: 1 }}
      />
      
      {/* Konva.js 层（上层，负责交互和控制） */}
      <Stage
        ref={konvaStageRef}
        width={stageWidth}
        height={stageHeight}
        className="absolute top-0 left-0"
        style={{ zIndex: 2, background: 'transparent' }}
      >
        <Layer>
          {/* Konva 只渲染透明的交互区域和控制手柄 */}
          {components.map(component => (
            <ComponentInteractionGroup key={component.id} component={component} />
          ))}
        </Layer>
      </Stage>
    </div>
  );
}
```

```typescript
  attribute vec2 aVertexPosition;

  uniform mat3 projectionMatrix;
  uniform mat3 translationMatrix;
  uniform mat3 uTextureMatrix;
  
  varying vec2 vTextureCoord;

export const fabricFragmentShader = `
  varying vec2 vTextureCoord;
  uniform sampler2D uSampler;
  uniform float uBrightness;
  uniform float uContrast;
  uniform float uSaturation;
  
  void main(void) {
    vec4 color = texture2D(uSampler, vTextureCoord);
    
    // 亮度调整
    color.rgb += uBrightness;
    
    // 对比度调整
    color.rgb = (color.rgb - 0.5) * uContrast + 0.5;
    
    // 饱和度调整
    float gray = dot(color.rgb, vec3(0.299, 0.587, 0.114));
    color.rgb = mix(vec3(gray), color.rgb, uSaturation);

  }
```

### 3.3 实时数字孪生 BOM 与成本引擎
#### 3.3.1 精确裁片面积计算算法
```typescript
// src/lib/bom/area-calculator.ts
import { Bezier } from 'bezier-js';

/**
 * 精确计算任意 SVG 路径的面积（使用格林公式）
 * @param pathData SVG 路径数据字符串
 * @returns 面积（平方毫米）
 */
export function calculatePathArea(pathData: string): number {
  // 解析 SVG 路径为点数组
  const points = parsePathToPoints(pathData, 100); // 每段曲线采样 100 个点
  
  // 使用格林公式计算多边形面积
  let area = 0;
  const n = points.length;
  
  for (let i = 0; i < n; i++) {
    const j = (i + 1) % n;
    area += points[i].x * points[j].y;
    area -= points[j].x * points[i].y;
  }
  
  return Math.abs(area / 2);
}

// 解析 SVG 路径为点数组
function parsePathToPoints(pathData: string, samplesPerCurve: number): { x: number; y: number }[] {
  const commands = pathData.match(/[a-zA-Z][^a-zA-Z]*/g) || [];
  const points: { x: number; y: number }[] = [];
  let currentX = 0;
  let currentY = 0;
  
  commands.forEach(cmd => {
    const type = cmd[0];
    const args = cmd.slice(1).trim().split(/[\s,]+/).map(parseFloat);
    
    switch (type) {
      case 'M':
        currentX = args[0];
        currentY = args[1];
        points.push({ x: currentX, y: currentY });
        break;
      case 'L':
        currentX = args[0];
        currentY = args[1];
        points.push({ x: currentX, y: currentY });
        break;
      case 'C':
        // 三次贝塞尔曲线
        const bezier = new Bezier(
          currentX, currentY,
          args[0], args[1],
          args[2], args[3],
          args[4], args[5]
        );
        
        for (let i = 1; i <= samplesPerCurve; i++) {
          const t = i / samplesPerCurve;
          const point = bezier.get(t);
          points.push({ x: point.x, y: point.y });
        }
        
        currentX = args[4];
        currentY = args[5];
        break;
      case 'Z':
        // 闭合路径
        if (points.length > 0) {
          points.push(points[0]);
        }
        break;
    }
  });
  
  return points;
}
```

#### 3.3.2 实时成本计算引擎
```typescript
// src/lib/bom/cost-engine.ts
import type { GeneInstance } from '@/types/gene';
import type { Material } from '@/types/material';
import { calculatePathArea } from './area-calculator';

export class CostEngine {
  private static instance: CostEngine;
  
  private constructor() {}
  
  static getInstance(): CostEngine {
    if (!CostEngine.instance) {
      CostEngine.instance = new CostEngine();
    }
    return CostEngine.instance;
  }

  /**
   * 计算单件成衣 FOB 成本
   */
  calculateFOBCost(
    geneInstances: GeneInstance[],
    materials: Map<string, Material>,
    parameters: {
      laborCost: number; // CMT 工费
      tariffRate: number; // 关税率
      exchangeRate: number; // 汇率
    }
  ): {
    totalCost: number;
    materialCost: number;
    laborCost: number;
    tariffCost: number;
    bomItems: BomItem[];
  } {
    const bomItems: BomItem[] = [];
    let materialCost = 0;

    // 计算每个基因实例的物料成本
    geneInstances.forEach(instance => {
      instance.paths.forEach(path => {
        if (!path.materialId) return;
        
        const material = materials.get(path.materialId);
        if (!material) return;
        
        // 计算裁片面积（平方毫米）
        const areaMm2 = calculatePathArea(path.data);
        const areaM2 = areaMm2 / 1000000;
        
        // 计算面料用量（米）
        const fabricWidthM = material.widthMm / 1000;
        const theoreticalUsage = areaM2 / fabricWidthM;
        const actualUsage = theoreticalUsage * (1 + material.lossRate);
        
        // 计算物料成本
        const itemCost = actualUsage * material.pricePerMeter;
        materialCost += itemCost;
        
        bomItems.push({
          id: `bom-${instance.id}-${path.id}`,
          componentName: `${instance.name} - ${path.id}`,
          material: material,
          usage: actualUsage,
          unit: 'm',
          cost: itemCost
        });
      });
    });

    // 计算总成本
    const laborCost = parameters.laborCost;
    const subtotal = materialCost + laborCost;
    const tariffCost = subtotal * parameters.tariffRate;
    const totalCost = (subtotal + tariffCost) * parameters.exchangeRate;

    return {
      totalCost: Math.round(totalCost * 100) / 100,
      materialCost: Math.round(materialCost * 100) / 100,
      laborCost: Math.round(laborCost * 100) / 100,
      tariffCost: Math.round(tariffCost * 100) / 100,
      bomItems
    };
  }
}
```

### 3.4 本地规则引擎
```typescript
// src/lib/rules/rule-engine.ts
import { evaluate } from 'mathjs';
import type { GeneInstance } from '@/types/gene';
import type { Rule } from '@/types/rule';

export class RuleEngine {
  private rules: Rule[] = [];

  loadRules(rules: Rule[]) {
    this.rules = rules;
  }

  /**
   * 校验基因实例是否符合所有规则
   */
  validateGeneInstance(instance: GeneInstance): RuleViolation[] {
    const violations: RuleViolation[] = [];
    
    this.rules.forEach(rule => {
      if (rule.applicableCategories && !rule.applicableCategories.includes(instance.gene.category)) {
        return;
      }
      
      const context = {
        params: instance.parameterValues,
        paths: instance.paths
      };
      
      try {
        const isValid = evaluate(rule.expression, context);
        if (!isValid) {
          violations.push({
            ruleId: rule.id,
            message: rule.message,
            level: rule.level
          });
        }
      } catch (e) {
        console.error(`Rule evaluation error: ${rule.id}`, e);
      }
    });
    
    return violations;
  }
}

export interface RuleViolation {
  ruleId: string;
  message: string;
  level: 'warning' | 'error';
}
```

```python
# backend/services/marker_service.py
import numpy as np
from shapely.geometry import Polygon
from shapely.ops import unary_union

class MarkerService:
    def calculate_marker_efficiency(self, pieces: list, fabric_width: int) -> dict:
        """
        计算排料利用率
        :param pieces: 裁片列表，每个裁片包含路径点数组
        :param fabric_width: 面料幅宽（毫米）
        :return: 排料结果，包含利用率和排料图
        """
        # 转换裁片为 Shapely 多边形
        polygons = []
        total_area = 0
        
        for piece in pieces:
            points = [(p['x'], p['y']) for p in piece['points']]
            poly = Polygon(points)
            polygons.append(poly)
            total_area += poly.area
        
        # 简单矩形排料算法（V2 阶段使用，V3 升级为启发式算法）
        current_x = 0
        current_y = 0
        max_height = 0
        placed_pieces = []
        
        for poly in polygons:
            bounds = poly.bounds
            width = bounds[2] - bounds[0]
            height = bounds[3] - bounds[1]
            
            if current_x + width > fabric_width:
                current_x = 0
                current_y += max_height
                max_height = 0
            
            placed_pieces.append({
                'polygon': poly,
                'x': current_x,
                'y': current_y
            })
            
            current_x += width
            if height > max_height:
                max_height = height
        
        # 计算排料面积和利用率
        marker_area = fabric_width * (current_y + max_height)
        efficiency = total_area / marker_area if marker_area > 0 else 0
        
        return {
            'efficiency': round(efficiency, 4),
            'marker_width': fabric_width,
            'marker_height': current_y + max_height,
            'total_area': total_area,
            'placed_pieces': [
                {
                    'id': piece['id'],
                    'x': p['x'],
                    'y': p['y']
                } for piece, p in zip(pieces, placed_pieces)
            ]
        }
```

```python
# backend/services/cost_service.py
from datetime import date
from sqlalchemy import and_

class CostService:
    def get_material_cost(self, material_code: str, origin: str, destination: str) -> dict:
        """
        获取物料的最新到岸成本
        """
        today = date.today()
        
        pricing = self.db.query(SupplyChainPricing).filter(
            and_(
                SupplyChainPricing.material_asset_code == material_code,
                SupplyChainPricing.origin_country == origin,
                SupplyChainPricing.destination_country == destination,
                SupplyChainPricing.effective_date <= today,
                (SupplyChainPricing.expiry_date >= today) | (SupplyChainPricing.expiry_date.is_(None))
            )
        ).order_by(SupplyChainPricing.effective_date.desc()).first()
        
        if not pricing:
            raise ValueError(f"No pricing found for material {material_code}")
        
        # 计算到岸单价
        landed_price = pricing.price_per_meter * (1 + pricing.tariff_rate)
        
        return {
            'material_code': material_code,
            'origin': origin,
            'destination': destination,
            'price_per_meter': pricing.price_per_meter,
            'tariff_rate': pricing.tariff_rate,
            'landed_price': landed_price,
            'currency': pricing.currency_code,
            'moq': pricing.moq,
            'lead_time_days': pricing.lead_time_days
        }
```
